# ================================================

# Server Configuration
PORT=5001
NODE_ENV=development  # development | production

# MongoDB Configuration
# For local MongoDB:
MONGO_URI=mongodb://127.0.0.1:27017/terra-cart
# For MongoDB Atlas:
# MONGO_URI=mongodb+srv://username:password@cluster.xxxxx.mongodb.net/terra-cart?retryWrites=true&w=majority

# ================================================
# SECURITY CONFIGURATION (IMPORTANT FOR PRODUCTION)
# ================================================

# JWT Secret - MUST be changed in production!
# Generate a strong secret: node -e "console.log(require('crypto').randomBytes(64).toString('hex'))"
JWT_SECRET=your-super-secure-jwt-secret-change-this-in-production

# Signed URL Secret (for secure file access)
# If not set, JWT_SECRET will be used as fallback
# Generate a unique secret for signed URLs
SIGNED_URL_SECRET=your-signed-url-secret-different-from-jwt

# API Base URL (for generating signed URLs)
# In production, set this to your actual domain
API_BASE_URL=http://localhost:5001

# ================================================
# CORS Configuration
# ================================================

# Allowed Origins (comma-separated)
# In production, set this to your actual frontend domains
# Example: https://admin.yourdomain.com,https://app.yourdomain.com
ALLOWED_ORIGINS=http://localhost:3000,http://localhost:5173,http://localhost:5174,http://localhost:5175

# ================================================
# FILE UPLOAD SECURITY
# ================================================

# Allow public uploads folder access (not recommended for production)
ALLOW_PUBLIC_UPLOADS=false

# Maximum file size in MB
MAX_FILE_SIZE=10

# ================================================
# RATE LIMITING
# ================================================

# Disable rate limiting entirely (development only - NOT recommended for production)
# Set to 'true' to disable all rate limiting in development
DISABLE_RATE_LIMIT=false

# Enable stricter rate limiting in production
# Values are requests per window (usually 15 minutes)
# In development, limits are automatically 10x higher
RATE_LIMIT_API=500
RATE_LIMIT_LOGIN=10
RATE_LIMIT_UPLOAD=50

# ================================================
# LOGGING
# ================================================

# Enable request logging (development only)
ENABLE_REQUEST_LOGGING=false

# ================================================
# ADDITIONAL SECURITY OPTIONS
# ================================================

# Enable HTTPS redirect (set to true when behind HTTPS proxy)
FORCE_HTTPS=false

# Session timeout in minutes
SESSION_TIMEOUT=60

# ================================================
# PRODUCTION CHECKLIST
# ================================================
# Before deploying to production, ensure:
# 
# 1. JWT_SECRET is changed from default
# 2. SIGNED_URL_SECRET is set
# 3. MONGO_URI points to production database
# 4. ALLOWED_ORIGINS is set to your frontend domains
# 5. NODE_ENV=production
# 6. ALLOW_PUBLIC_UPLOADS=false
# 7. API_BASE_URL is set to your domain
# 8. SSL/TLS is enabled (handled by reverse proxy)
# 9. Database backups are configured
